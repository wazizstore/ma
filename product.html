// ── نظام Google Sheets ────────────────────────────────────────── const ORDER_ENDPOINT = 'رابط_Web_App_حقك_هنا'; // استبدل بالرابط let cart = JSON.parse(sessionStorage.getItem('wz_cart') || '[]'); function updateBadge() { document.getElementById('cartBadge').textContent = cart.length; } function showToast(msg, type = '') { const t = document.getElementById('toast'); t.textContent = msg; t.className = type; t.style.display = 'block'; setTimeout(() => t.style.display = 'none', 3000); } function renderShop() { const grid = document.getElementById('shopGrid'); grid.innerHTML = products.map((p, i) => { const disc = Math.round((1 - p.price / p.originalPrice) * 100); return `
${disc > 0 ? `-${disc}%` : ''}
${p.name}
${p.price} درهم ${p.originalPrice} درهم
`; }).join(''); } function toggleCart() { document.getElementById('cartDrawer').classList.toggle('open'); document.getElementById('cartOverlay').classList.toggle('open'); renderCart(); } function renderCart() { const body = document.getElementById('cartBody'); const foot = document.getElementById('cartFoot'); if (!cart.length) { body.innerHTML = `
السلة فارغة
`; foot.style.display = 'none'; return; } let total = 0; body.innerHTML = cart.map((item, idx) => { total += parseInt(item.price); return `
${item.name}
${(item.color || item.size) ? `
${[item.color, item.size].filter(Boolean).join(' · ')}
` : ''}
${item.price} درهم
`; }).join(''); document.getElementById('cartTotal').textContent = total + ' درهم'; foot.style.display = 'block'; } function rmCart(idx) { cart.splice(idx, 1); sessionStorage.setItem('wz_cart', JSON.stringify(cart)); updateBadge(); renderCart(); } function openCheckout() { document.getElementById('coDrawer').classList.add('open'); document.getElementById('coOverlay').classList.add('open'); } function closeCheckout() { document.getElementById('coDrawer').classList.remove('open'); document.getElementById('coOverlay').classList.remove('open'); } async function submitCart() { const name = document.getElementById('cName').value.trim(); const phone = document.getElementById('cPhone').value.trim(); const city = document.getElementById('cCity').value.trim(); const addr = document.getElementById('cAddr').value.trim(); if (!name || !phone || !city || !addr) { showToast('يرجى ملء جميع الحقول', 'error'); return; } if (!cart.length) { showToast('السلة فارغة', 'error'); return; } const btn = document.getElementById('cartSubmitBtn'); btn.disabled = true; btn.innerHTML = 'جاري الإرسال... '; try { const response = await fetch(ORDER_ENDPOINT, { method: 'POST', headers: { 'Content-Type': 'application/json', }, body: JSON.stringify({ name: name, phone: phone, city: city, address: addr, items: cart.map(item => ({ name: item.name, color: item.color || '', size: item.size || '', price: item.price })) }) }); const result = await response.json(); if (result.status === 'success') { document.getElementById('coBody').innerHTML = `

تم استلام طلبك بنجاح!

شكراً ${name}، سنتصل بك قريباً لتأكيد طلبك.

${result.orderId ? `

رقم الطلب: ${result.orderId}

` : ''}
`; cart = []; sessionStorage.setItem('wz_cart', JSON.stringify(cart)); updateBadge(); } else { throw new Error(result.message || 'فشل إرسال الطلب'); } } catch (error) { console.error('خطأ في إرسال الطلب:', error); showToast('حدث خطأ، حاول مرة أخرى: ' + error.message, 'error'); btn.disabled = false; btn.innerHTML = ' تأكيد الطلب الآن'; } } document.addEventListener('DOMContentLoaded', () => { updateBadge(); renderShop(); });
